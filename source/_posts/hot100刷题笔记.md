## 快速排序和归并排序模板


```cpp
void quick_sort(int q[], int l, int r)
{
    if (l >= r) return; //数组中没有元素或只有一个元素递归停止

    int i = l - 1, j = r + 1, x = q[l + r >> 1];
    while (i < j)
        {
            do i ++ ; while (q[i] < x);     
            do j -- ; while (q[j] > x);
            if (i < j) swap(q[i], q[j]);        
        }
    quick_sort(q, l, j), quick_sort(q, j + 1, r);
}
```

## 堆的存储 

一般都使用数组来表示堆，（若根节点从0开始，因为数组的初始下标就是0），i结点的父结点的下标就是(i-1)/2，i结点的的左右结点下标分别为2* i+1,2*i+2。  

如果数组初始下标为1，则i结点的父结点的下标就是i/2，i结点的的左右结点下标分别为2* i,2*i+1。  

题目：

输入一个长度为n的整数数列，从小到大输出前m小的数。

**输入格式**

第一行包含整数n和m。

第二行包含n个整数，表示整数数列。

**输出格式**

共一行，包含m个整数，表示整数数列中前m小的数。

**数据范围**

1≤m≤n≤10^5 ， 1≤数列中元素≤10^9

```cpp
#include <iostream>
#include <algorithm>
using namespace std;

const int N = 1e5 + 10;

int n, m;
int h[N], size;

void down(int u) {
    int t = u;
    // 经过以下两步，t变为u, 2u, 2u+1 中最小的节点的编号
    if(2 * u <= size && h[t] > h[2 * u]) t = u * 2;
    if(2 * u + 1 <= size && h[t] > h[2 * u + 1]) t = u * 2 + 1;
    
    // 如果u不是最小的，则和最小的值交换，然后递归处理
    if(t != u) {
        swap(h[u], h[t]);
        down(t);
    }
}

void up(int u) {
    while (u / 2 && h[u / 2] > h[u]) {
        swap(h[u / 2], h[u]);
        u /= 2;
    }
}

int main() {
    scanf("%d%d", &n, &m);
    for(int i = 1; i <= n; i ++ ) scanf("%d", &h[i]);
    size = n;
    
    // 创建堆，复杂度为O(n)
    for(int i = n / 2; i; i -- ) down(i);
    
    while (m --) {
        print("%d ", h[1]);
        h[1] = h[size];
        size --;
        down(1);
    }
    
    return 0;
}


```

## 215. 数组中的第K个最大元素  （大根堆、小根堆、升序排列、降序排列）

**示例 2:**

```plain

输出: 4
```

+ `-10<sup>4</sup> <= nums[i] <= 10<sup>4</sup>`

```cpp
class Solution {
public:
    int findKthLargest(vector<int>& nums, int k) {
        int n = nums.size();
        return quickselect(nums, 0, n - 1, k);
    }
    int quickselect(vector<int>& nums,int l, int r, int k){
         if (l == r)
            return nums[k - 1];
        int i = l - 1, j = r + 1;
        int p = nums[l];
        while(i < j)
        {
            do i++; while(nums[i] > p);
            do j--; while(nums[j] < p);
            if(i < j)
            swap(nums[i], nums[j]);
        }
        if (k - 1 <= j) {
        return quickselect(nums, l, j, k);  // 继续查找左半部分
    } else {
        return quickselect(nums, j + 1, r, k);  // 继续查找右半部分
    }
        

    }
};
```

建立k个小根堆，使用向上调整建堆，向下调整维护堆（偏序排序）

```cpp
class Solution {
      // 自定义小根堆
        vector<int> heap;
public:
    int findKthLargest(vector<int>& nums, int k) {
        

        // 向小根堆中插入元素
        for (int i = 0; i < nums.size(); i++) {
            if (heap.size() < k) {
                heap.push_back(nums[i]);  // 堆不满，直接插入
                heapifyUp(heap.size() - 1);  // 调整堆
            } else if (nums[i] > heap[0]) {
                heap[0] = nums[i];  // 替换堆顶元素
                heapifyDown(0);  // 调整堆
            }
        }

        return heap[0];  // 堆顶元素即为第k大的元素
    }

private:
    // 向上调整堆
    void heapifyUp(int index) {
        while (index > 0) {
            int parent = (index - 1) / 2;
            if (heap[index] < heap[parent]) {
                swap(heap[index], heap[parent]);
                index = parent;
            } else {
                break;
            }
        }
    }

    // 向下调整堆
    void heapifyDown(int index) {
        int leftChild, rightChild, smallest;
        while (2 * index + 1 < heap.size()) {
            leftChild = 2 * index + 1;
            rightChild = 2 * index + 2;
            smallest = index;

            if (heap[leftChild] < heap[smallest]) {
                smallest = leftChild;
            }
            if (rightChild < heap.size() && heap[rightChild] < heap[smallest]) {
                smallest = rightChild;
            }
            if (smallest != index) {
                swap(heap[index], heap[smallest]);
                index = smallest;
            } else {
                break;
            }
        }
    }
};
```

先把数组全部元素建立大根堆，然后再一个个pick堆顶元素，知道第k个（全序排序）: 删除k-1次，然后最后剩下的堆的堆顶就是第k个最大元素。

```cpp
class Solution {
public:
    void maxHeapify(vector<int>& a, int i, int heapSize) {
        int l = i * 2 + 1, r = i * 2 + 2, largest = i;
        if (l < heapSize && a[l] > a[largest]) {
            largest = l;
        } 
        if (r < heapSize && a[r] > a[largest]) {
            largest = r;
        }
        if (largest != i) {
            swap(a[i], a[largest]);
            maxHeapify(a, largest, heapSize);
        }
    }

    void buildMaxHeap(vector<int>& a, int heapSize) {
        for (int i = heapSize / 2 - 1; i >= 0; --i) {
            maxHeapify(a, i, heapSize);
        } 
    }

    int findKthLargest(vector<int>& nums, int k) {
        int heapSize = nums.size();
        buildMaxHeap(nums, heapSize);
        for (int i = nums.size() - 1; i >= nums.size() - k + 1; --i) {
            swap(nums[0], nums[i]);
            --heapSize;
            maxHeapify(nums, 0, heapSize);
        }
        return nums[0];
    }
};

```

```cpp
class Solution {
public:
    int maxArea(vector<int>& height) {
       int left = 0;
       int right = height.size() - 1; 
       int ans = 0;
       while(left < right)
       {
        ans = max(ans, min(height[left], height[right])* (right - left));
        if(height[left] < height[right]) left ++;
        else right --;
       }
       return ans;
    }
};
```

```cpp
class Solution {

        }
```

## 20.有效括号（哈希表or not）
给定一个只包括 `'('`，`')'`，`'{'`，`'}'`，`'['`，`']'` 的字符串 `s` ，判断字符串是否有效。

有效字符串需满足：

1. 左括号必须用相同类型的右括号闭合。	
2. 左括号必须以正确的顺序闭合。	
3. 每个右括号都有一个对应的相同类型的左括号。

**示例 1：**

**输入：**s = "()"

**示例 2：**

**示例 3：**

**提示：**

+ `1 <= s.length <= 10<sup>4</sup>`
+ `s` 仅由括号 `'()[]{}'` 组成

```cpp
class Solution {
public:
    bool isValid(string s) {
       stack<char> st;
       for(char c : s)
       {
        if(c == '(')
        {
            st.push(')');
        }else if(c == '{')
        {
            st.push('}');
        }else if(c == '[')
        {
            st.push(']');
        }else{
          if(st.empty() || c != st.top())
          {
            return false;
          }
          st.pop();
        }
       }
       return st.empty();
    }
};
```

```cpp
    int largestRectangleArea(vector<int>& heights) {
    int n = heights.size();
    vector<int> left(n, -1);
    stack<int> stk;
    for(int i = 0; i < n; i++)
    {
        while(!stk.empty() && heights[i] <= heights[stk.top()])
        {
            stk.pop();
        }
        if(!stk.empty())

        }
        stk.push(i);
    }

    for(int i = 0; i < n; i++)
    {
       ans = max(ans, heights[i] * (right[i] - left[i] - 1)); 
    }

    }
```

```cpp
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}

class Solution {

    ListNode* mergeTwoLists(ListNode* list1, ListNode* list2) {

        return dummyhead->next;      
    }
```

```cpp
       if(i == n) 
       {
        ans.emplace_back(path);
        return;
       }

       {
        if(!on_path[j])

            on_path[j] = true;
```

```cpp
    // 1. 成员变量：记录中序遍历的前一个节点值（前驱节点值）
    long long pre = LLONG_MIN;

public:
    bool isValidBST(TreeNode* root) {
        // 2. 递归终止条件：遍历到空节点，返回合法
        if (root == nullptr) {
            return true;
        }

        // 3. 中序遍历：先递归验证左子树（左）
        if (!isValidBST(root->left)) { // 左子树验证失败，直接返回false

        if (root->val <= pre) { // 当前节点值 <= 前驱节点值，违反严格递增，返回false
            return false;
        }

        // 5. 更新前驱节点值：当前节点成为下一个节点的前驱
        pre = root->val;

        // 6. 中序遍历：递归验证右子树（右）
        return isValidBST(root->right); // 右子树的验证结果即为当前子树的验证结果
```

+ `s` 中所有整数的取值范围为 `[1, 300]` 

```cpp
class Solution {
public:
    string decodeString(string s) {
        if (s.empty()) {
            return s;
        }

        // s[0] 是字母
        if (isalpha(s[0])) {
            // 分离出 s[0]，解码剩下的
            return s[0] + decodeString(s.substr(1));
        }

        // s[0] 是数字，后面至少有一对括号
        int i = s.find('['); // 找左括号
        // 找右括号，注意对于 [...[...]...] 这种情况，第一个右括号并不是我们要找的，第二个才是
        int balance = 1; // 左括号个数减去右括号个数
        for (int j = i + 1; ; j++) {
            if (s[j] == '[') {
                balance++;
            } else if (s[j] == ']') {
                balance--;
                if (balance == 0) { // 左右括号个数相等，找到与 s[i] 匹配的右括号 s[j]
                    int k = stoi(s.substr(0, i));
                    string t = decodeString(s.substr(i + 1, j - i - 1));
                    string res;
                    while (k--) {
                        res += t;
                    }
                    return res + decodeString(s.substr(j + 1));
                }
            }
        }
    }
};
```

```cpp
class MedianFinder {
priority_queue<int> left;
priority_queue<int, vector<int>, greater<>> right;
public:

void addNum(int num) {
    if(left.size() == right.size()){
        right.push(num);
        left.push(right.top());
        right.pop();
    }
    else{
        left.push(num);
        right.push(left.top());
        left.pop();
    }
}

double findMedian() {
    if(left.size() > right.size())
    {
        return left.top();
    }
    else{
        return (left.top() + right.top())/2.0;
    }
}
};
```

```cpp
        if(node == nullptr || k == 0){
            return;
        }
        dfs(node->left, k, ans);
        if(--k == 0)
        {
            ans = node->val;
        }
        dfs(node->right, k, ans);
    }

        int ans;
        dfs(root, k, ans);

    }
```

```cpp
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    vector<int> rightSideView(TreeNode* root) {
        vector<int> res;
        if(root == nullptr) return res; // 根为空直接返回
        
        queue<TreeNode*> que;
        que.push(root); // 只入队非空根节点
        
        while(!que.empty()){
            int levelSize = que.size(); // 每层节点数，核心分层逻辑
            for(int i = 0; i < levelSize; i++){
                TreeNode* cur = que.front();
                que.pop();
                // 每层最后一个节点，加入右视图结果
                if(i == levelSize - 1){
                    res.push_back(cur->val);
                }
                // ✅ 核心修复：严格判空后入队，绝对不推入nullptr
                if(cur->left != nullptr) que.push(cur->left);
                if(cur->right != nullptr) que.push(cur->right);
            }
        }
        return res;
    }
};
```

```cpp
           int n = preorder.size();
        unordered_map<int, int> idx_map;
        for (int i = 0; i < n; ++i) {
            idx_map[inorder[i]] = i;
        }
        return dfs(0, n, 0, preorder, inorder, idx_map);
    }
};
```

<font style="color:rgb(31, 35, 41);background-color:rgba(0, 0, 0, 0);">用的是</font>**<font style="color:rgb(31, 35, 41);background-color:rgba(0, 0, 0, 0);">左闭右开区间</font>**<font style="color:rgb(31, 35, 41);background-color:rgba(0, 0, 0, 0);">，这是这个题目的最优区间写法，优势是：</font>

+ <font style="color:rgb(31, 35, 41);background-color:rgba(0, 0, 0, 0);">区间的长度 = </font>`<font style="color:rgb(31, 35, 41);background-color:rgba(0, 0, 0, 0);">pre_r - pre_l</font>`<font style="color:rgb(31, 35, 41);background-color:rgba(0, 0, 0, 0);">，无需计算 </font>`<font style="color:rgb(31, 35, 41);background-color:rgba(0, 0, 0, 0);">+1/-1</font>`<font style="color:rgb(31, 35, 41);background-color:rgba(0, 0, 0, 0);">，直接得到节点数量；</font>
+ <font style="color:rgb(31, 35, 41);background-color:rgba(0, 0, 0, 0);">终止条件极简：</font>`<font style="color:rgb(31, 35, 41);background-color:rgba(0, 0, 0, 0);">pre_l == pre_r</font>`<font style="color:rgb(31, 35, 41);background-color:rgba(0, 0, 0, 0);"> 就代表区间为空，返回空节点即可；</font>
+ <font style="color:rgb(31, 35, 41);background-color:rgba(0, 0, 0, 0);">所有递归的边界划分都不用纠结下标偏移，不易出错。</font>

 把整段递归代码的逻辑浓缩成三句话，**<font style="color:rgb(31, 35, 41);background-color:rgba(0, 0, 0, 0);">这三句话就是前序 + 中序构建二叉树的全部精髓</font>**：  

+ **<font style="color:rgb(31, 35, 41);background-color:rgba(0, 0, 0, 0);">找根节点</font>**<font style="color:rgb(31, 35, 41);background-color:rgba(0, 0, 0, 0);">：通过</font>`<font style="color:rgb(31, 35, 41);background-color:rgba(0, 0, 0, 0);">preorder[pre_l]</font>`<font style="color:rgb(31, 35, 41);background-color:rgba(0, 0, 0, 0);">确定当前子树的根节点值；</font>

+ **<font style="color:rgb(31, 35, 41);background-color:rgba(0, 0, 0, 0);">组装树</font>**<font style="color:rgb(31, 35, 41);background-color:rgba(0, 0, 0, 0);">：执行</font>`<font style="color:rgb(31, 35, 41);background-color:rgba(0, 0, 0, 0);">return new TreeNode(...)</font>`<font style="color:rgb(31, 35, 41);background-color:rgba(0, 0, 0, 0);">，把左右子树挂载到根节点上，返回根节点指针。</font>

```cpp
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode(int x) : val(x), left(NULL), right(NULL) {}
 * };

class Solution {
public:
    TreeNode* lowestCommonAncestor(TreeNode* root, TreeNode* p, TreeNode* q) {
         // 递归终止条件：空节点 或 找到p/q，直接返回当前节点
        if (root == nullptr || root == p || root == q) {
            return root;
        }
        // 后序遍历：先遍历左子树，找p/q的踪迹
        TreeNode* left = lowestCommonAncestor(root->left, p, q);
        // 后序遍历：再遍历右子树，找p/q的踪迹
        TreeNode* right = lowestCommonAncestor(root->right, p, q);
        
        // 场景1：左右子树都找到，当前节点是最近公共祖先
        if (left && right) {
            return root;
        }
        // 场景2/3/4：传递结果，有则返回找到的节点，无则返回空
        return left ? left : right; 
    }
};
```

## 437.路径总和III（回溯）
给定一个二叉树的根节点 `root` ，和一个整数 `targetSum` ，求该二叉树里节点值之和等于 `targetSum` 的 **路径** 的数目。

**路径** 不需要从根节点开始，也不需要在叶子节点结束，但是路径方向必须是向下的（只能从父节点到子节点）。

**示例 1：**

![](https://cdn.nlark.com/yuque/0/2026/jpeg/39118916/1768021274632-342d74d9-846d-47fa-ac9c-fa4e2e75ce82.jpeg)

```plain
输入：root = [10,5,-3,3,2,null,11,3,-2,null,1], targetSum = 8
输出：3
解释：和等于 8 的路径有 3 条，如图所示。
```

**示例 2：**

```plain
输入：root = [5,4,8,11,null,13,4,7,2,null,null,5,1], targetSum = 22
输出：3
```

**提示:**

+ 二叉树的节点个数的范围是 `[0,1000]`
+ `-10<sup>9</sup> <= Node.val <= 10<sup>9</sup>` 
+ `-1000 <= targetSum <= 1000`

```cpp
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
     int res = 0;
    int target;
    unordered_map<long long, int> preSumMap; // 前缀和:出现次数

    void dfs(TreeNode* root, long long curSum) {
        if (root == nullptr) return;

        // 1. 先序遍历：访问当前节点，更新当前前缀和
        curSum += root->val;

        // 2. 核心：统计符合条件的路径数 → preSum[j-1] = curSum - target
        res += preSumMap[curSum - target];
    
        // 把当前前缀和加入哈希表
        preSumMap[curSum]++;

        // 3. 递归遍历左右子树
        dfs(root->left, curSum);
        dfs(root->right, curSum);

        // 4. 回溯：恢复前缀和和哈希表 → 核心！！！
        preSumMap[curSum]--;
    }
public:
    int pathSum(TreeNode* root, int targetSum) {
        target = targetSum;
        preSumMap[0] = 1; // 初始化：前缀和为0出现1次，处理从根节点开始的路径
        dfs(root, 0);
        return res;
    }
};
```

注意 二叉树的**<font style="color:rgb(31, 35, 41);background-color:rgba(0, 0, 0, 0);">普通先序遍历一维数组是「扁平化」的</font>**，这个数组丢失了二叉树的 **<font style="color:rgb(31, 35, 41);background-color:rgba(0, 0, 0, 0);">「路径分支 / 层级信息」</font>** ！而本题的核心规则是：**<font style="color:rgb(31, 35, 41);background-color:rgba(0, 0, 0, 0);">合法路径只能是「父→子 向下走」，不能跨分支、不能回溯、不能走兄弟节点</font>**。

+ `-1000 <= Node.val <= 1000`

```cpp
class Solution {
public:
    int maxPathSum(TreeNode* root) {
        // 1. 初始化全局最大值为INT_MIN（处理全负数的情况，如所有节点都是-1）
        int ans = INT_MIN;
        // 2. 定义递归DFS函数（C++20的this auto&&实现递归lambda）
        // 捕获&：引用外部变量ans；参数：当前节点node；返回值：int（当前节点的最大链和）
        auto dfs = [&](this auto&& dfs, TreeNode* node) -> int {
            // 3. 递归终止条件：空节点的链和为0（无贡献）
            if (node == nullptr) {
                return 0;
            }
            // 4. 后序遍历：先递归计算左、右子树的最大链和（自底向上）
            int l_val = dfs(node->left);   // 左子树的最大链和
            int r_val = dfs(node->right);  // 右子树的最大链和
            
            // 5. 核心1：更新全局最大值ans
            // 当前节点作为「路径最高点」的路径和 = 左链和 + 节点值 + 右链和
            ans = max(ans, l_val + r_val + node->val);
            
            // 6. 核心2：返回当前节点的最大链和（只能选左/右其一，保证路径连续）
            // max(l_val, r_val) + node->val：选左/右中更大的链和，加上当前节点值
            // 和0取max：如果链和为负，不如不选该子树（链和为0），避免拉低父节点的链和
            return max(max(l_val, r_val) + node->val, 0);
        };
        // 7. 从根节点开始DFS
        dfs(root);
        // 8. 返回全局最大值
        return ans;
    }
};
```

## 146.LRU缓存（双向链表）
请你设计并实现一个满足  [LRU (最近最少使用) 缓存](https://baike.baidu.com/item/LRU) 约束的数据结构。

实现 `LRUCache` 类：

+ `LRUCache(int capacity)` 以 **正整数** 作为容量 `capacity` 初始化 LRU 缓存
+ `int get(int key)` 如果关键字 `key` 存在于缓存中，则返回关键字的值，否则返回 `-1` 。
+ `void put(int key, int value)` 如果关键字 `key` 已经存在，则变更其数据值 `value` ；如果不存在，则向缓存中插入该组 `key-value` 。如果插入操作导致关键字数量超过 `capacity` ，则应该 **逐出** 最久未使用的关键字。

函数 `get` 和 `put` 必须以 `O(1)` 的平均时间复杂度运行。

**示例：**

```plain
输入
["LRUCache", "put", "put", "get", "put", "get", "put", "get", "get", "get"]
[[2], [1, 1], [2, 2], [1], [3, 3], [2], [4, 4], [1], [3], [4]]
输出
[null, null, null, 1, null, -1, null, -1, 3, 4]

解释
LRUCache lRUCache = new LRUCache(2);
lRUCache.put(1, 1); // 缓存是 {1=1}
lRUCache.put(2, 2); // 缓存是 {1=1, 2=2}
lRUCache.get(1);    // 返回 1
lRUCache.put(3, 3); // 该操作会使得关键字 2 作废，缓存是 {1=1, 3=3}
lRUCache.get(2);    // 返回 -1 (未找到)
lRUCache.put(4, 4); // 该操作会使得关键字 1 作废，缓存是 {4=4, 3=3}
lRUCache.get(1);    // 返回 -1 (未找到)
lRUCache.get(3);    // 返回 3
lRUCache.get(4);    // 返回 4
```

**提示：**

+ `1 <= capacity <= 3000`
+ `0 <= key <= 10000`
+ `0 <= value <= 10<sup>5</sup>`
+ 最多调用 `2 * 10<sup>5</sup>` 次 `get` 和 `put`

```cpp
    // 双向链表节点结构
struct Node {
    int key;
    int value;
    Node* prev;
    Node* next;
    Node(int k = 0, int v = 0) : key(k), value(v), prev(nullptr), next(nullptr) {}
};
class LRUCache {
 unordered_map<int, Node*> keyToNode; // 哈希表：key -> 节点指针，O(1)查找
    int cap;                             // 缓存最大容量
    Node* dummy;       
     void addToHead(Node* node) {
        node->prev = dummy;
        node->next = dummy->next;
        dummy->next->prev = node;
        dummy->next = node;
    }
    // ② 删除链表中的任意指定节点 
    void removeNode(Node* node) {
        node->prev->next = node->next;
        node->next->prev = node->prev;
    }

    // ③ 把节点移动到链表头部 = 先删后加 
    void moveToHead(Node* node) {
        removeNode(node);
        addToHead(node);
    }

    // ④ 删除链表【尾部节点】(dummy的前驱，最久未使用的位置)，并返回该节点
    Node* removeTail() {
        Node* lastNode = dummy->prev; // 
        removeNode(lastNode);
        return lastNode;
    }


public:
    LRUCache(int capacity) {
        this->cap = capacity;
        dummy = new Node();           // 创建唯一的哨兵节点
        dummy->prev = dummy;          // 哨兵前驱指向自己
        dummy->next = dummy;          // 哨兵后继指向自己 → 形成【自循环的空双向链表】 
    }
    
    int get(int key) {
       if (keyToNode.find(key) == keyToNode.end()) {
            return -1;
        }
        Node* node = keyToNode[key];
        moveToHead(node);
        return node->value; 
    }
    
    void put(int key, int value) {
       // 情况1：key存在 → 更新值+移到头部
        if (keyToNode.find(key) != keyToNode.end()) {
            Node* node = keyToNode[key];
            node->value = value;
            moveToHead(node);
            return;
        }

        // 情况2：key不存在 → 新增节点
        Node* newNode = new Node(key, value);
        keyToNode[key] = newNode;
        addToHead(newNode);

        // 情况3：容量超限 → 淘汰最久未使用的尾部节点
        if (keyToNode.size() > cap) {
            Node* delNode = removeTail();
            keyToNode.erase(delNode->key);
            delete delNode; // 释放内存，避免内存泄漏
        } 
    }
};

/**
 * Your LRUCache object will be instantiated and called as such:
 * LRUCache* obj = new LRUCache(capacity);
 * int param_1 = obj->get(key);
 * obj->put(key,value);
 */
```

```cpp
        for(int i = 0; i < m; i++){
            for(int j = 0; j < n; j++){
                if(grid[i][j] == 1){
                    fresh ++;
                }else if(grid[i][j] == 2){
                    q.emplace_back(i, j);
                }
            } 
        }
        int ans = 0;
        while(fresh && !q.empty()){
            ans++;
            vector<pair<int, int>> nxt;
            for(auto& [x, y] :q){
                for(auto d : directions){
                    int i = x + d[0], j = y +d[1];
                    if(i >= 0 && i < m && j >= 0&& j < n && grid[i][j] == 1){
                        fresh --;
                        grid[i][j] = 2;
                        nxt.emplace_back(i, j);
                    }
                }
            }
            q = move(nxt);
        }
```

```cpp
class Solution {
public:
    vector<vector<string>> solveNQueens(int n) {
        // 1. 存储最终结果：所有合法的棋盘布局（每个布局是n行字符串）
        vector<vector<string>> ans;
        // 2. 初始化n×n棋盘：所有位置为'.'（空，无皇后）
        vector board(n, string(n, '.')); // 等价于 vector<string> board(n, string(n, '.'))
        // 3. 状态数组：标记列、主对角线、副对角线是否被皇后占用（uint8_t比vector<bool>高效）
        vector<bool> col(n), diag1(n * 2 - 1), diag2(n * 2 - 1);
       // col[c] = true → 第c列有皇后
         // 主对角线（\）：r+c为定值，共2n-1条
        // 副对角线（/）：r-c为定值，转换为非负索引后共2n-1条
        
        // 4. 递归DFS函数（C++20特性：this auto&& dfs 实现递归lambda）
        // 捕获方式：& 捕获所有外部变量（ans/board/col/diag1/diag2/n），r是当前处理的行号
        auto dfs = [&](this auto&& dfs, int r) {
            // 5. 递归终止条件：所有行（0~n-1）都放好皇后
            if (r == n) {
                ans.push_back(board); // 复制当前棋盘到结果集（找到一个合法布局）
                return;
            }
            // 6. 遍历当前行的所有列，尝试放置皇后
            for (int c = 0; c < n; c++) {
                // 7. 副对角线索引转换：r-c的范围是-(n-1)~n-1，加n-1转为非负（0~2n-2）
                int rc = r - c + n - 1;
                // 8. 冲突判断：当前列、主对角线、副对角线都无皇后 → 可以放置
                if (!col[c] && !diag1[r + c] && !diag2[rc]) {
                    // 9. 放置皇后：当前位置(r,c)设为'Q'
                    board[r][c] = 'Q';
                    // 10. 标记状态：当前列、主对角线、副对角线被占用
                    col[c] = diag1[r + c] = diag2[rc] = true;
                    // 11. 递归处理下一行（放置下一个皇后）
                    dfs(r + 1);
                    // 12. 恢复现场（回溯核心）：递归返回后，撤销当前行的皇后和状态标记
                    col[c] = diag1[r + c] = diag2[rc] = false;
                    board[r][c] = '.';
                }
            }
        };
        // 13. 从第0行开始递归（逐行放置皇后）
        dfs(0);
        // 14. 返回所有合法布局
        return ans;
    }
};
```

+ 你能将算法的时间复杂度降低到 `O(n log(n))` 吗?

```cpp
class Solution {
public:
    int lengthOfLIS(vector<int>& nums) {
        // 1. 获取数组长度n（元素个数）
        int n = nums.size();
        // 2. 定义DP数组f：f[i] = 以nums[i]为最后一个元素的最长递增子序列长度
        vector<int> f(n);
        // 3. 遍历每个元素i，计算f[i]
        for (int i = 0; i < n; i++) {
            // 4. 初始化f[i]为0：后续找j的最大值后，需要+1（包含自己）
            f[i] = 0;
            // 5. 遍历i之前的所有元素j（j < i），找能接在j后面的最长子序列
            for (int j = 0; j < i; j++) {
                // 6. 关键条件：nums[j] < nums[i] → nums[i]可以接在nums[j]的子序列后面
                if (nums[j] < nums[i]) {
                    // 7. 取j中能形成的最长长度（更新f[i]为最大的f[j]）
                    f[i] = max(f[i], f[j]);
                }
            }
            // 8. 加上自己：子序列至少包含nums[i]本身，所以长度+1
            f[i]++;
        }
        // 9. 返回f数组的最大值：整个数组的LIS长度（LIS可以以任意元素结尾）
        return ranges::max(f);
    }
};
```

```cpp
class Solution {
public:
    bool canPartition(vector<int>& nums) {
    int s = reduce(nums.begin(), nums.end());
    if(s % 2) return false;
    s /= 2;
    vector<int> f(s + 1, false);
    f[0] = true;
    int s2 = 0;
    for(int x : nums){
        s2 = min(s2 + x, s);
        for(int j = s2; j >= x; j--){
            f[j] = f[j] || f[j - x];
        }
        
        if(f[s]) return true;
    }
    return false;
    }
};
```

**示例 3：**

+ `s[i]` 为 `'('` 或 `')'`

```cpp
class Solution {
public:
    int longestValidParentheses(string s) {
        stack<int> stk;
        stk.push(-1); // 虚拟红线（哨兵）：作为有效括号的初始左边界，解决边界问题
        int ans = 0;  // 记录最长有效括号长度
        // 遍历每个字符，i是当前字符的下标
        for (int i = 0; i < s.size(); i++) {
            if (s[i] == '(') { // 遇到左括号（待匹配，代码里比喻为“炸弹”）
                stk.push(i);   // 压入左括号的下标，标记为“待匹配的位置”
            } else if (stk.size() > 1) { // 遇到右括号，且栈中不止哨兵（有可匹配的左括号）
                stk.pop(); // 弹出栈顶的左括号下标（完成匹配，“拆弹”）
                // 计算当前有效长度：i - 新栈顶（最近的未匹配位置）
                ans = max(ans, i - stk.top());
            } else { // 遇到右括号，但栈中只有哨兵（无匹配的左括号，无法拆弹）
                stk.top() = i; // 替换哨兵为当前下标，作为新的“无效左边界”
            }
        }
        return ans;
    }
};
```

## 5.最长回文子串
给你一个字符串 `s`，找到 `s` 中最长的回文子串 。

**示例 1：**

```plain
输入：s = "babad"
输出："bab"
解释："aba" 同样是符合题意的答案。
```

**示例 2：**

```plain
输入：s = "cbbd"
输出："bb"
```

**提示：**	

+ `1 <= s.length <= 1000`
+ `s` 仅由数字和英文字母组成

```cpp
class Solution {
public:
    string longestPalindrome(string s) {
        // 1. 获取字符串长度n
        int n = s.size();
        // 2. 初始化最长回文串的左右边界（左闭右闭区间）：初始为第一个字符（长度1）
        int ans_left = 0, ans_right = 0;

        // 3. 遍历所有2n-1个回文中心（i从0到2n-2）
        for (int i = 0; i < 2 * n - 1; i++) {
            // 4. 计算当前中心对应的左右扩展起点l和r
            int l = i / 2, r = (i + 1) / 2;
            // 5. 中心扩展：向左右扩散，直到字符不相等或越界
            while (l >= 0 && r < n && s[l] == s[r]) {
                l--; // 左指针左移
                r++; // 右指针右移
            }
            // 6. 计算当前扩展得到的回文串长度（循环结束后，有效区间是l+1到r-1）
            // 有效长度 = (r-1) - (l+1) + 1 = r - l - 1
            if (r - l - 1 > ans_right - ans_left) {
                // 7. 更新最长回文串的边界（左闭右闭）
                ans_left = l + 1;
                ans_right = r - 1;
            }
        }

        // 8. 截取并返回最长回文子串：substr(起始下标, 长度)
        return s.substr(ans_left, ans_right - ans_left + 1);
    }
};
```

```plain

输出：0
解释：两个字符串没有公共子序列，返回 0 。
```

+ `1 <= text1.length, text2.length <= 1000`
+ `text1` 和 `text2` 仅由小写英文字符组成。

```cpp
class Solution {
public:
    int longestCommonSubsequence(string s, string t) {
        // 1. 获取字符串t的长度m（将t作为一维数组的维度，节省空间）
        int m = t.size();
        // 2. 定义一维DP数组f：f[j] 对应二维DP的 dp[i-1][j]（上一轮的状态），长度m+1（包含j=0的边界）
        vector<int> f(m + 1);
        // 3. 遍历s的每个字符x（对应二维DP的i从1到s.size()）
        for (char x : s) {
            // 4. 初始化pre=0：对应二维DP的 dp[i-1][0] = 0（左上角初始状态），j=0时的上一轮值
            for (int j = 0, pre = 0; j < m; j++) {
                // 5. 保存当前f[j+1]到tmp：这是「上一轮j+1的状态」（dp[i-1][j+1]），后续会被覆盖，需要先保存
                int tmp = f[j + 1];
                // 6. 核心状态转移：
                // - 若x == t[j]（当前字符相等）：f[j+1] = pre + 1（pre是dp[i-1][j]，即左上角）
                // - 若x != t[j]（当前字符不等）：f[j+1] = max(f[j+1]（上一轮j+1，即正上方）, f[j]（本轮j，即正左方）)
                f[j + 1] = x == t[j] ? pre + 1 : max(f[j + 1], f[j]);
                // 7. 更新pre为tmp：pre变成下一轮的「左上角」（dp[i-1][j+1]）
                pre = tmp;
            }
        }
        // 8. 返回f[m]：对应二维DP的 dp[s.size()][m]，即两个完整字符串的LCS长度
        return f[m];
    }
};
```

## 72.编辑距离
给你两个单词 `word1` 和 `word2`， _请返回将 _`_word1_`_ 转换成 _`_word2_`_ 所使用的最少操作数_  。

你可以对一个单词进行如下三种操作：

+ 替换一个字符

**示例 1：**


**示例 2：**

```plain
输入：word1 = "intention", word2 = "execution"
输出：5
解释：
intention -> inention (删除 't')
inention -> enention (将 'i' 替换为 'e')

exection -> execution (插入 'u')
```

+ `word1` 和 `word2` 由小写英文字母组成

```cpp
class Solution {
public:
    int minDistance(string s, string t) {
        // 1. 获取字符串t的长度m（将t作为一维数组的维度，节省空间）
        int m = t.length();
        // 2. 定义一维DP数组f：f[j] 对应二维DP的 dp[i-1][j]（上一轮的状态），长度m+1（包含j=0的边界）
        vector<int> f(m + 1);
        // 3. 初始化边界：dp[0][j] = j（空串转t的前j个字符，需要插入j次）
        for (int j = 0; j < m; j++) {
            f[j + 1] = j + 1; // f[1]=1, f[2]=2,...f[m]=m，对应dp[0][1]=1, dp[0][2]=2...
        }
        // 4. 遍历s的每个字符x（对应二维DP的i从1到s.size()）
        for (char x : s) {
            // 5. 保存上一轮的f[0]（dp[i-1][0]）到pre：作为第一个j的左上角状态
            int pre = f[0];
            // 6. 更新f[0]：对应dp[i][0] = i（s的前i个字符转空串，需要删除i次）
            f[0]++; // 每遍历一个s的字符，i+1，因此f[0]++
            // 7. 遍历t的每个字符t[j]（对应二维DP的j从1到m）
            for (int j = 0; j < m; j++) {
                // 8. 保存当前f[j+1]到tmp：这是「上一轮j+1的状态」（dp[i-1][j+1]），后续会被覆盖，需要先保存
                int tmp = f[j + 1];
                // 9. 核心状态转移：
                // - 若x == t[j]（当前字符相等）：f[j+1] = pre（dp[i-1][j]，无需操作）
                // - 若x != t[j]（当前字符不等）：f[j+1] = min(正上方、正左方、左上角) + 1（三种操作选最少的）
                f[j + 1] = x == t[j] ? pre : min({f[j + 1], f[j], pre}) + 1;
                // 10. 更新pre为tmp：pre变成下一轮j+1的「左上角」（dp[i-1][j+1]）
                pre = tmp;
            }
        }
        // 11. 返回f[m]：对应二维DP的 dp[s.size()][m]，即s转t的最小操作数
        return f[m];
    }
};
```

```cpp
        vector f(n + 1, vector<int>(m + 1));

            f[0][j + 1] = j + 1;
        }
        for (int i = 0; i < n; i++) {
            f[i + 1][0] = i + 1;
            for (int j = 0; j < m; j++) {
```
